```yaml
name: Generate Snake Animation
on:
  schedule:
    - cron: "0 0 * * *"
  workflow_dispatch:
permissions:
  contents: write

  generate:
    runs-on: ubuntu-latest
    steps:
      - name: Generate Snake

          github_user_name: JeebanBeura

          target_branch: output
          build_dir: dist
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
```
